import bank from './src/data/careerQuestions/v7/index.js';

const CLASS_KEY = 'graduate';

/*
|--------------------------------------------------------------------------
| Resolve Graduate question bank safely
|--------------------------------------------------------------------------
*/

function resolveGraduateQuestions(source) {
  const directCandidates = [
    source?.[CLASS_KEY],
    source?.graduate,
    source?.GRADUATE,

    source?.questions?.[CLASS_KEY],
    source?.questions?.graduate,
    source?.questions?.GRADUATE,

    source?.classes?.[CLASS_KEY],
    source?.classes?.graduate,
    source?.classes?.GRADUATE,

    source?.questionBank?.[CLASS_KEY],
    source?.questionBank?.graduate,
    source?.questionBank?.GRADUATE,

    source?.careerQuestions?.[CLASS_KEY],
    source?.careerQuestions?.graduate,
    source?.careerQuestions?.GRADUATE,
  ];

  for (const candidate of directCandidates) {
    if (
      Array.isArray(candidate) &&
      candidate.length > 0
    ) {
      return candidate;
    }
  }

  /*
  |--------------------------------------------------------------------------
  | If default export itself is already an array
  |--------------------------------------------------------------------------
  */

  if (Array.isArray(source)) {
    /*
    | First try to identify Graduate rows explicitly.
    */

    const graduateRows =
      source.filter(question => {
        const classValue =
          question?.classKey ??
          question?.class ??
          question?.studentClass ??
          question?.level ??
          question?.className ??
          '';

        return (
          String(classValue)
            .trim()
            .toLowerCase() ===
          CLASS_KEY
        );
      });

    if (graduateRows.length > 0) {
      return graduateRows;
    }

    /*
    | If the imported array itself contains exactly the
    | Graduate bank, use it directly.
    */

    if (source.length === 500) {
      return source;
    }
  }

  /*
  |--------------------------------------------------------------------------
  | Search one level deeper for an array that looks like the
  | Graduate question collection.
  |--------------------------------------------------------------------------
  */

  if (
    source &&
    typeof source === 'object'
  ) {
    for (const value of Object.values(source)) {
      if (!Array.isArray(value)) {
        continue;
      }

      if (value.length === 500) {
        return value;
      }

      const graduateRows =
        value.filter(question => {
          if (
            !question ||
            typeof question !== 'object'
          ) {
            return false;
          }

          const classValue =
            question.classKey ??
            question.class ??
            question.studentClass ??
            question.level ??
            question.className ??
            '';

          return (
            String(classValue)
              .trim()
              .toLowerCase() ===
            CLASS_KEY
          );
        });

      if (graduateRows.length > 0) {
        return graduateRows;
      }
    }
  }

  return [];
}

/*
|--------------------------------------------------------------------------
| Utilities
|--------------------------------------------------------------------------
*/

function countBy(items, getter) {
  const map = new Map();

  for (const item of items) {
    const value = getter(item);

    if (Array.isArray(value)) {
      for (const rawValue of value) {
        const key =
          String(rawValue || '')
            .trim() ||
          '(empty)';

        map.set(
          key,
          (map.get(key) || 0) + 1
        );
      }

      continue;
    }

    const key =
      String(value || '')
        .trim() ||
      '(empty)';

    map.set(
      key,
      (map.get(key) || 0) + 1
    );
  }

  return [...map.entries()]
    .sort(
      (a, b) =>
        b[1] - a[1]
    );
}

function printDistribution(
  title,
  rows,
  total
) {
  console.log('');
  console.log(title);
  console.log('-'.repeat(60));

  for (
    const [name, count]
    of rows
  ) {
    const pct =
      total > 0
        ? (
            (count / total) *
            100
          ).toFixed(1)
        : '0.0';

    console.log(
      `${String(name).padEnd(32)} ${String(count).padStart(4)}  ${pct.padStart(6)}%`
    );
  }
}

function hasValues(value) {
  return (
    Array.isArray(value) &&
    value.length > 0
  );
}

    !Array.isArray(bank)
  ) {
    console.error(
      'Top-level bank keys:',
      Object.keys(bank)
    );
  }

  process.exit(1);
}

/*
|--------------------------------------------------------------------------
| Core questions
|--------------------------------------------------------------------------
|
| Your previous audit reported:
|
| Potential core: 32
|
| Some banks may not explicitly use core/isCore.
| So we support explicit markers first.
|--------------------------------------------------------------------------
*/

const explicitCore =
  questions.filter(
    question =>
      question.core === true ||
      question.isCore === true
  );

const unrestrictedCore =
  questions.filter(
    question => {
      const hasDegree =
        hasValues(
          question.degrees
        );

      const hasSpecialization =
        hasValues(
          question.specializations
        );

      const hasStatus =
        hasValues(
          question.statuses
        );

      const hasExperience =
        hasValues(
          question.experiences ??
          question.experienceLevels
        );

      const hasGoal =
        hasValues(
          question.goals
        );

      const hasSkill =
        hasValues(
          question.skills
        );

      return (
        !hasDegree &&
        !hasSpecialization &&
        !hasStatus &&
        !hasExperience &&
        !hasGoal &&
        !hasSkill
      );
    }
  );

const core =
  explicitCore.length > 0
    ? explicitCore
    : unrestrictedCore;
