import {
  redisGetJson,
  redisSetJson,
  redisEnabled,
} from '../services/redisCache.js';


/*
|--------------------------------------------------------------------------
| COLLEGE REDIS CACHE
|--------------------------------------------------------------------------
*/

const CATALOG_CACHE_KEY =
  'cw:college:catalog:v2';


const DETAIL_CACHE_PREFIX =
  'cw:college:detail:v2:';


const CATALOG_TTL_SECONDS =
  Math.max(
    60,
    Number(
      process.env
        .COLLEGE_CATALOG_CACHE_TTL_SECONDS ||
      900
    )
  );


const DETAIL_TTL_SECONDS =
  Math.max(
    60,
    Number(
      process.env
        .COLLEGE_DETAIL_CACHE_TTL_SECONDS ||
      1800
    )
  );


/*
|--------------------------------------------------------------------------
| QUERY CHECK
|--------------------------------------------------------------------------
*/

function hasQueryParameters(
  req
) {

  return (
    req.query &&
    Object.keys(
      req.query
    ).length > 0
  );
}


/*
|--------------------------------------------------------------------------
| BUILD CACHE CONFIG
|--------------------------------------------------------------------------
*/

function getCacheConfig(
  req
) {

  /*
  |--------------------------------------------------------------------------
  | Full catalog
  |--------------------------------------------------------------------------
  |
  | GET /api/colleges
  |
  |--------------------------------------------------------------------------
  */

  if (
    req.method ===
      'GET' &&
    req.path ===
      '/' &&
    !hasQueryParameters(
      req
    )
  ) {

    return {
      key:
        CATALOG_CACHE_KEY,

      ttl:
        CATALOG_TTL_SECONDS,

      type:
        'catalog',
    };
  }


  /*
  |--------------------------------------------------------------------------
  | College detail
  |--------------------------------------------------------------------------
  |
  | Examples:
  |
  | GET /api/colleges/123
  |
  | We intentionally avoid caching special non-detail routes.
  |
  |--------------------------------------------------------------------------
  */

  if (
    req.method ===
      'GET' &&
    /^\/[^/]+$/.test(
      req.path
    )
  ) {

    const segment =
      req.path
        .slice(1)
        .trim();


    const reserved =
      new Set([
        'search',
        'catalog',
        'states',
        'types',
        'featured',
        'health',
      ]);


    if (
      segment &&
      !reserved.has(
        segment.toLowerCase()
      )
    ) {

      return {
        key:
          DETAIL_CACHE_PREFIX +
          encodeURIComponent(
            segment
          ),

        ttl:
          DETAIL_TTL_SECONDS,

        type:
          'detail',
      };
    }
  }


  return null;
}


/*
|--------------------------------------------------------------------------
| CACHE MIDDLEWARE
|--------------------------------------------------------------------------
*/

export async function collegeRedisCache(
  req,
  res,
  next
) {

  try {

    const config =
      getCacheConfig(
        req
      );


    if (!config) {
      return next();
    }


    /*
    |--------------------------------------------------------------------------
    | Redis unavailable/disabled
    |--------------------------------------------------------------------------
    */

    if (!redisEnabled()) {

      res.set(
        'X-CW-Redis-Cache',
        'DISABLED'
      );


      return next();
    }


    /*
    |--------------------------------------------------------------------------
    | Try Redis first
    |--------------------------------------------------------------------------
    */

    const cached =
      await redisGetJson(
        config.key
      );


    if (
      cached !==
      null
    ) {

      res.set(
        'X-CW-Redis-Cache',
        'HIT'
      );


      res.set(
        'X-CW-Cache-Type',
        config.type
      );


      return res.json(
        cached
      );
    }


    /*
    |--------------------------------------------------------------------------
    | Cache miss
    |--------------------------------------------------------------------------
    */

    res.set(
      'X-CW-Redis-Cache',
      'MISS'
    );


    /*
    |--------------------------------------------------------------------------
    | Capture original res.json
    |--------------------------------------------------------------------------
    */

    const originalJson =
      res.json.bind(
        res
      );


    let alreadySent =
      false;


    res.json =
      function cachedJson(
        body
      ) {

        if (alreadySent) {

          return originalJson(
            body
          );
        }


        alreadySent =
          true;


        /*
        |--------------------------------------------------------------------------
        | Only cache successful GET responses
        |--------------------------------------------------------------------------
        */

        if (
          res.statusCode >= 200 &&
          res.statusCode < 300
        ) {

          /*
          |--------------------------------------------------------------------------
          | Do not block response waiting for Redis write
          |--------------------------------------------------------------------------
          */

          redisSetJson(
            config.key,
            body,
            config.ttl
          )
            .catch(
              (error) => {

                console.warn(
                  '[COLLEGE REDIS CACHE] write failed:',
                  error?.message ||
                  error
                );
              }
            );
        }


        return originalJson(
          body
        );
      };


    return next();

  } catch (error) {

    /*
    |--------------------------------------------------------------------------
    | FAIL OPEN
    |--------------------------------------------------------------------------
    |
    | Cache failure must never break college APIs.
    |
    |--------------------------------------------------------------------------
    */

    console.warn(
      '[COLLEGE REDIS CACHE] middleware failed:',
      error?.message ||
      error
    );


    return next();
  }
}


export default collegeRedisCache;
